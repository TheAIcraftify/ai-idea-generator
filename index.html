const form = document.getElementById("ideaForm"); const businessType = document.getElementById("businessType"); const problem = document.getElementById("problem"); const customers = document.getElementById("customers"); const loadingPanel = document.getElementById("loadingPanel"); const resultsPanel = document.getElementById("resultsPanel"); const resSolution = document.getElementById("resSolution"); const resFeatures = document.getElementById("resFeatures"); const resAudience = document.getElementById("resAudience"); const resTech = document.getElementById("resTech"); const resEarning = document.getElementById("resEarning"); const resRoadmap = document.getElementById("resRoadmap"); const regenerateBtn = document.getElementById("regenerateBtn"); const newIdeaBtn = document.getElementById("newIdeaBtn"); function makeIdea(type, problemText, customerText) { const typeLower = type.toLowerCase(); let solution = `A smart digital solution for ${type} that helps solve "${problemText}" and makes the business easier to manage and grow.`; let features = [ `Simple dashboard designed for ${type}`, `Easy management of the main business activities`, `Customer-friendly interface and quick actions`, `Automatic reminders and follow-ups`, `Simple reports to track business performance` ]; let technology = [ "HTML, CSS and JavaScript", "Responsive design for mobile and desktop", "Local storage for basic data saving", "GitHub Pages for free hosting" ]; let earning = [ "Offer the solution as a paid service to businesses", "Freemium model with optional premium features", "Monthly subscription for advanced tools" ]; let roadmap = [ `Define the main workflow for the ${type} business`, "Build the basic interface in HTML and CSS", "Add JavaScript functionality", "Test the solution with real business examples", "Add data storage and advanced features", "Deploy the working version online" ]; if ( typeLower.includes("restaurant") || typeLower.includes("cafe") || typeLower.includes("food") || typeLower.includes("bakery") || typeLower.includes("catering") ) { features = [ "Digital menu and product management", "Order and customer management", "Order status and follow-up system", "Daily sales summary", "Customer history and repeat-order tracking" ]; } else if ( typeLower.includes("school") || typeLower.includes("academy") || typeLower.includes("tutor") || typeLower.includes("education") || typeLower.includes("coaching") ) { features = [ "Student and teacher management", "Attendance tracking", "Class and timetable management", "Fee and payment reminders", "Progress and performance reports" ]; } else if ( typeLower.includes("hospital") || typeLower.includes("clinic") || typeLower.includes("doctor") || typeLower.includes("medical") ) { features = [ "Patient information management", "Appointment scheduling", "Automatic appointment reminders", "Patient history", "Simple daily activity dashboard" ]; } else if ( typeLower.includes("salon") || typeLower.includes("beauty") || typeLower.includes("spa") || typeLower.includes("barber") ) { features = [ "Appointment booking", "Customer profile management", "Service and price management", "Automatic appointment reminders", "Sales and customer history" ]; } else if ( typeLower.includes("shop") || typeLower.includes("store") || typeLower.includes("retail") ) { features = [ "Product and inventory management", "Customer management", "Sales tracking", "Low-stock reminders", "Daily and monthly sales reports" ]; } else if ( typeLower.includes("real estate") || typeLower.includes("property") ) { features = [ "Property listing management", "Customer inquiry tracking", "Property search and filtering", "Viewing appointment management", "Lead and follow-up reminders" ]; } else if ( typeLower.includes("travel") || typeLower.includes("tour") || typeLower.includes("tourism") ) { features = [ "Trip and package management", "Customer booking management", "Travel itinerary creation", "Payment and booking tracking", "Customer follow-up reminders" ]; } else if ( typeLower.includes("gym") || typeLower.includes("fitness") ) { features = [ "Member management", "Membership expiry reminders", "Workout and progress tracking", "Payment tracking", "Member attendance history" ]; } else if ( typeLower.includes("freelance") || typeLower.includes("agency") || typeLower.includes("consult") ) { features = [ "Client management", "Project and task tracking", "Deadline reminders", "Invoice and payment tracking", "Client communication history" ]; } return { solution, features, audience: `Designed for ${customerText || "customers and users of this business"} who regularly interact with ${type}.`, technology, earning, roadmap }; } function showList(element, items) { element.innerHTML = ""; items.forEach(item => { const li = document.createElement("li"); li.textContent = item; element.appendChild(li); }); } function generateIdea() { const type = businessType.value.trim(); const problemText = problem.value.trim(); const customerText = customers.value.trim(); if (!type || !problemText || !customerText) { return; } loadingPanel.hidden = false; resultsPanel.hidden = true; setTimeout(() => { const idea = makeIdea(type, problemText, customerText); resSolution.textContent = idea.solution; resAudience.textContent = idea.audience; showList(resFeatures, idea.features); showList(resTech, idea.technology); showList(resEarning, idea.earning); showList(resRoadmap, idea.roadmap); loadingPanel.hidden = true; resultsPanel.hidden = false; resultsPanel.scrollIntoView({ behavior: "smooth", block: "start" }); }, 700); } form.addEventListener("submit", function(event) { event.preventDefault(); generateIdea(); }); regenerateBtn.addEventListener("click", function() { generateIdea(); }); newIdeaBtn.addEventListener("click", function() { form.reset(); resultsPanel.hidden = true; loadingPanel.hidden = true; window.scrollTo({ top: 0, behavior: "smooth" }); });
